local function fetch(url)
	if type(request) == "function" then
		local ok, body = pcall(request, { Url = url, Method = "GET" })
		if ok and type(body) == "string" and #body > 0 then return body end
	end
	if type(http_request) == "function" then
		local ok, body = pcall(http_request, url)
		if ok and type(body) == "string" and #body > 0 then return body end
	end
	local ok, body = pcall(function() return game:HttpGet(url) end)
	if ok and type(body) == "string" and #body > 0 then return body end
	return nil
end

local source = fetch(PAYLOAD_URL)
if not source then error("[Aero] no se pudo descargar el payload") end

local chunk, err
if type(loadstring) == "function" then chunk, err = loadstring(source) end
if not chunk and type(load) == "function" then chunk, err = load(source) end
if not chunk then error("[Aero] el payload no compila: " .. tostring(err)) end
